Polling the cloud every " + String(POLL_INTERVAL_MS / 1000) + "s."); } /* ---------------- Main loop ---------------- */ void loop() { if (WiFi.status() != WL_CONNECTED) { connectWiFi(); } if (millis() - lastPoll >= POLL_INTERVAL_MS) { lastPoll = millis(); pollServer(); } if (!manualOverride) { checkSchedules(); } } /* ---------------- WiFi ---------------- */ void connectWiFi() { if (WiFi.status() == WL_CONNECTED) return; Serial.print("Connecting to WiFi"); WiFi.mode(WIFI_STA); WiFi.begin(WIFI_SSID, WIFI_PASS); int attempts = 0; while (WiFi.status() != WL_CONNECTED && attempts < 40) { delay(500); Serial.print("."); attempts++; } Serial.println(); if (WiFi.status() == WL_CONNECTED) { Serial.println("WiFi connected: " + WiFi.localIP().toString()); } else { Serial.println("WiFi connect failed — will retry in loop()."); } } /* ---------------- Cloud communication ---------------- */ String httpsPost(const String& url, const String& body) { WiFiClientSecure client; client.setInsecure(); // skips TLS cert validation — fine for a hobby project talking to Google's own domain, // but note this doesn't verify you're really talking to script.google.com HTTPClient http; if (!http.begin(client, url)) return ""; http.addHeader("Content-Type", "text/plain"); // Apps Script reads e.postData.contents regardless of content-type int code = http.POST(body); String response = ""; if (code > 0) response = http.getString(); else Serial.println("POST failed, code: " + String(code)); http.end(); return response; } String httpsGet(const String& url) { WiFiClientSecure client; client.setInsecure(); HTTPClient http; if (!http.begin(client, url)) return ""; int code = http.GET(); String response = ""; if (code > 0) response = http.getString(); else Serial.println("GET failed, code: " + String(code)); http.end(); return response; } void pollServer() { if (WiFi.status() != WL_CONNECTED) return; StaticJsonDocument<256> reqDoc; reqDoc["key"] = SHARED_KEY; reqDoc["action"] = "poll"; reqDoc["relayState"] = ledOn ? "ON" : "OFF"; reqDoc["uptime"] = millis() / 1000; reqDoc["vcc"] = readBatteryVoltage(); reqDoc["wifiRSSI"] = WiFi.RSSI(); reqDoc["firmwareVersion"] = FIRMWARE_VERSION; String body; serializeJson(reqDoc, body); String response = httpsPost(GAS_URL, body); if (response.length() == 0) { Serial.println("Poll: no response."); return; } StaticJsonDocument<512> respDoc; DeserializationError err = deserializeJson(respDoc, response); if (serverSchedulesVersion != schedulesVersion) { fetchSchedules(serverSchedulesVersion); } const char* serverTime = respDoc["serverTime"]; if (serverTime) syncTimeFromServerString(serverTime); const char* command = respDoc["command"]; if (command && strlen(command) > 0 && strcmp(command, "REBOOT") == 0) { addLog("Cloud requested reboot"); delay(300); ESP.restart(); } const char* otaUrl = respDoc["otaUrl"]; const char* otaVersion = respDoc["otaVersion"]; if (otaUrl && strlen(otaUrl) > 0 && otaVersion && strcmp(otaVersion, FIRMWARE_VERSION) != 0) { performOta(otaUrl); } } void applyDesiredState(const String& state) { if (state == "ON") { if (!ledOn) { ledOn = true; digitalWrite(LED_PIN, HIGH); addLog("Cloud: Manual ON"); } manualOverride = true; } else if (state == "OFF") { if (ledOn) { ledOn = false; digitalWrite(LED_PIN, LOW); addLog("Cloud: Manual OFF"); } manualOverride = true; } else { // AUTO if (manualOverride) addLog("Cloud: Back to AUTO (schedule control)"); manualOverride = false; } } void fetchSchedules(long newVersion) { String url = String(GAS_URL) + "?key=" + SHARED_KEY; if (err) { Serial.println("Fetch schedules: JSON parse error."); return; } JsonArray arr = doc["schedules"].as(); numSchedules = 0; for (JsonObject item : arr) { if (numSchedules >= MAX_SCHEDULES) break; Schedule &s = schedules[numSchedules]; strncpy(s.date, item["date"] | "", sizeof(s.date) - 1); strncpy(s.onTime, item["onTime"] | "", sizeof(s.onTime) - 1); strncpy(s.offTime, item["offTime"] | "", sizeof(s.offTime) - 1); s.repeat = item["repeat"] | false; s.startPercent = item["startPercent"] | 0; s.targetPercent = item["targetPercent"] | 0; s.minutesPerPercent = item["minutesPerPercent"] | 0; numSchedules++; } schedulesVersion = newVersion; saveSchedulesToFlash(); addLog("Schedules synced (" + String(numSchedules) + " total)"); } /* ---------------- Time ---------------- The Sheet's server time (always accurate, no local NTP needed) is applied on every successful poll. TIME_OFFSET_SECONDS shifts it to local time so the hour()/minute()-style helpers below read local wall-clock time directly. */ void syncTimeFromServerString(const char* iso) { int y, mo, d, h, mi, s; if (sscanf(iso, "%d-%d-%dT%d:%d:%d", &y, &mo, &d, &h, &mi, &s) != 6) return; struct tm tmStruct = {}; tmStruct.tm_year = y - 1900; tmStruct.tm_mon = mo - 1; tmStruct.tm_mday = d; tmStruct.tm_hour = h; tmStruct.tm_min = mi; tmStruct.tm_sec = s; time_t utcEpoch = mktime(&tmStruct); // no TZ configured, so this is treated as UTC time_t localEpoch = utcEpoch + TIME_OFFSET_SECONDS; struct timeval tv = { localEpoch, 0 }; settimeofday(&tv, nullptr); } int nowHour() { time_t t = time(nullptr); return localtime(&t)->tm_hour; } int nowMinute() { time_t t = time(nullptr); return localtime(&t)->tm_min; } int nowSecond() { time_t t = time(nullptr); return localtime(&t)->tm_sec; } int nowDay() { time_t t = time(nullptr); return localtime(&t)->tm_mday; } int nowMonth() { time_t t = time(nullptr); return localtime(&t)->tm_mon + 1; } int nowYear() { time_t t = time(nullptr); return localtime(&t)->tm_year + 1900; } /* ---------------- Schedule evaluation (same logic as the original sketch) ---------------- */ int timeToMinutes(const char* t) { if (strlen(t) < 5) return 0; int h = (t[0] - '0') * 10 + (t[1] - '0'); int m = (t[3] - '0') * 10 + (t[4] - '0'); return h * 60 + m; } int calculateBatteryPercent(int startPercent, int minutesPerPercent, int onMin) { int nowMin = nowHour() * 60 + nowMinute(); int elapsed = nowMin - onMin; if (elapsed < 0) elapsed = 0; int percentIncrease = (minutesPerPercent > 0) ? elapsed / minutesPerPercent : 0; return startPercent + percentIncrease; } void checkSchedules() { if (nowYear() < 2020) return; // time not synced yet int nowMin = nowHour() * 60 + nowMinute(); if (skipUntilMin != -1) { if (nowMin < skipUntilMin) return; else skipUntilMin = -1; } for (int i = 0; i < numSchedules; i++) { Schedule &s = schedules[i]; bool ok = s.repeat; if (!s.repeat) { int sy = atoi(s.date); int sm = atoi(s.date + 5); int sd = atoi(s.date + 8); ok = (nowYear() == sy && nowMonth() == sm && nowDay() == sd); } if (!ok) continue; int onMin = timeToMinutes(s.onTime); int offMin = timeToMinutes(s.offTime); bool shouldBeOn = (onMin < offMin) ? (nowMin >= onMin && nowMin < offMin) : (nowMin >= onMin || nowMin < offMin); if (shouldBeOn && !ledOn) { ledOn = true; digitalWrite(LED_PIN, HIGH); addLog("Schedule ON"); } if (!shouldBeOn && ledOn) { ledOn = false; digitalWrite(LED_PIN, LOW); addLog("Schedule OFF"); } } } /* ---------------- Logging (sent straight to the Sheet) ---------------- */ void addLog(const String& msg) { Serial.println(msg); if (WiFi.status() != WL_CONNECTED) return; StaticJsonDocument<256> doc; doc["key"] = SHARED_KEY; doc["action"] = "log"; doc["message"] = msg; String body; serializeJson(doc, body); httpsPost(GAS_URL, body); // fire-and-forget; response isn't needed } /* ---------------- Battery voltage ---------------- Only meaningful if you've wired a resistor divider from a battery/supply into VOLTAGE_PIN. If you don't need this, this can just return 0 — it's sent along with every poll purely for display on the control page. */ float readBatteryVoltage() { int raw = analogRead(VOLTAGE_PIN); float pinVoltage = (raw / 4095.0) * 3.3; // rough — ESP32 ADC is non-linear near the rails return pinVoltage * VOLTAGE_DIVIDER_RATIO; } /* ---------------- OTA (firmware update over the internet) ---------------- */ void performOta(const char* url) { addLog("Starting OTA update from cloud..."); delay(200); // give the log POST time to leave before we start overwriting flash WiFiClientSecure client; client.setInsecure(); httpUpdate.rebootOnUpdate(true); t_httpUpdate_return ret = httpUpdate.update(client, url); switch (ret) { case HTTP_UPDATE_FAILED: addLog("OTA failed: " + httpUpdate.getLastErrorString()); break; case HTTP_UPDATE_NO_UPDATES: addLog("OTA: server reported no update needed"); break; case HTTP_UPDATE_OK: addLog("OTA succeeded, rebooting"); // device reboots automatically; this may not even send break; } } /* ---------------- Flash storage (Preferences / NVS) ---------------- */ void saveSchedulesToFlash() { prefs.begin("relay", false); prefs.putInt("numSched", numSchedules); prefs.putBytes("schedules", schedules, sizeof(schedules)); prefs.putLong("schedVer", schedulesVersion); prefs.end(); } void loadSchedulesFromFlash() { prefs.begin("relay", true); numSchedules = prefs.getInt("numSched", 0); if (numSchedules < 0 || numSchedules > MAX_SCHEDULES) numSchedules = 0; prefs.getBytes("schedules", schedules, sizeof(schedules)); schedulesVersion = prefs.getLong("schedVer", -1); prefs.end(); }
